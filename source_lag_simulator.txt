    <script>
        let isRunning = false;
        let animationId = null;
        let startTime = null;
        let frameCount = 0;
        let lastFrameTime = performance.now();
        let fps = 60;

        const fpsDisplay = document.getElementById('fpsDisplay');
        const timeElapsed = document.getElementById('timeElapsed');
        const lagFactorDisplay = document.getElementById('lagFactor');
        const frameTimeDisplay = document.getElementById('frameTime');
        const operationsDisplay = document.getElementById('operations');
        const lagFill = document.getElementById('lagFill');
        const startBtn = document.getElementById('startBtn');
        const stopBtn = document.getElementById('stopBtn');

        function updateDisplay(currentTime) {
            const elapsedTime = (currentTime - startTime) / 1000;
            const deltaTime = currentTime - lastFrameTime;
            lastFrameTime = currentTime;

            frameCount++;

            // Calculate FPS every 10 frames
            if (frameCount % 10 === 0) {
                fps = Math.round(1000 / deltaTime);
                timeElapsed.textContent = elapsedTime.toFixed(1) + 's';
                lagFactorDisplay.textContent = confettiParticles.length.toLocaleString();
                frameTimeDisplay.textContent = deltaTime.toFixed(2) + 'ms';
                const n = confettiParticles.length;
                operationsDisplay.textContent = (Math.min(n * 2, 500)).toLocaleString();

                // Update FPS display color
                fpsDisplay.className = 'fps-display';
                if (fps >= 50) fpsDisplay.classList.add('good');
                else if (fps >= 30) fpsDisplay.classList.add('ok');
                else if (fps >= 15) fpsDisplay.classList.add('bad');
                else fpsDisplay.classList.add('terrible');

                fpsDisplay.textContent = fps + ' FPS';

                // Update lag bar — based on actual FPS (60 = 0%, 0 = 100%)
                const lagPercentage = Math.min(Math.max((1 - fps / 60) * 100, 0), 100);
                lagFill.style.width = lagPercentage + '%';
            }
        }

        function gameLoop(currentTime) {
            if (!isRunning) return;

            const elapsedTime = (startTime ? (currentTime - startTime) / 1000 : 0);

            // Heavy physics + matrix math
            heavyPhysics();

            // Spawn confetti — rate grows with time
            const confettiBatch = Math.floor(elapsedTime * 3) + 5;
            spawnConfettiBatch(confettiBatch);
            if (elapsedTime > 5) currentThemeIdx++;  // cycle themes
            drawConfetti();

            updateDisplay(currentTime);

            animationId = requestAnimationFrame(gameLoop);
        }

        function startSimulation() {
            if (isRunning) return;

            isRunning = true;
            startTime = performance.now();
            lastFrameTime = startTime;
            frameCount = 0;

            startBtn.style.display = 'none';
            stopBtn.style.display = 'inline-block';

            animationId = requestAnimationFrame(gameLoop);
        }

        function stopSimulation() {
            isRunning = false;
            if (animationId) {
                cancelAnimationFrame(animationId);
                animationId = null;
            }

            startBtn.style.display = 'inline-block';
            stopBtn.style.display = 'none';

            frameCount = 0;
            fps = 60;
            startTime = null;
            lastFrameTime = performance.now();

            confettiParticles = [];
            confettiCtx.clearRect(0, 0, confettiCanvas.width, confettiCanvas.height);
            currentThemeIdx = 0;

            fpsDisplay.className = 'fps-display good';
            fpsDisplay.textContent = '60 FPS';
            timeElapsed.textContent = '0s';
            lagFactorDisplay.textContent = '0';
            frameTimeDisplay.textContent = '16.67ms';
            operationsDisplay.textContent = '0';
            lagFill.style.width = '0%';
        }

        function resetSimulation() {
            stopSimulation();
        }

        // ── CONFETTI SYSTEM (from guessing game) ─────────────────────────────
        const confettiCanvas = document.getElementById('confetti-canvas');
        const confettiCtx = confettiCanvas.getContext('2d');
        confettiCanvas.width = window.innerWidth;
        confettiCanvas.height = window.innerHeight;
        window.addEventListener('resize', () => {
            confettiCanvas.width = window.innerWidth;
            confettiCanvas.height = window.innerHeight;
        });

        let confettiParticles = [];
        const confettiThemes = {
            rainbow: () => `hsl(${Math.random() * 360}, 70%, 50%)`,
            gold:    () => `hsl(${45 + Math.random() * 15}, 90%, ${50 + Math.random() * 20}%)`,
            fire:    () => `hsl(${10 + Math.random() * 30}, 100%, ${50 + Math.random() * 30}%)`,
        };
        let currentThemeIdx = 0;
        const themeKeys = Object.keys(confettiThemes);

        const GRAVITY = 0.9;
        const BOUNCINESS = 0.55;
        const AIR_FRICTION = 0.992;
        const TRAIL_LENGTH = 8;

        function spawnConfettiBatch(count) {
            const colorFunc = confettiThemes[themeKeys[currentThemeIdx % themeKeys.length]];
            for (let i = 0; i < count; i++) {
                const size = Math.random() * 12 + 4;
                confettiParticles.push({
                    x: Math.random() * confettiCanvas.width,
                    y: -size,
                    size,
                    mass: size * size * 0.01 + Math.random() * 0.5,
                    speedX: (Math.random() - 0.5) * 6,
                    speedY: Math.random() * 2 + 1,
                    rotation: Math.random() * 360,
                    rotationSpeed: Math.random() * 12 - 6,
                    angularDamping: 0.97 + Math.random() * 0.02,
                    color: colorFunc(),
                    bounceCount: 0,
                    trail: [],
                });
            }
        }

        function drawConfetti() {
            confettiCtx.clearRect(0, 0, confettiCanvas.width, confettiCanvas.height);
            const W = confettiCanvas.width, H = confettiCanvas.height;

            for (let i = confettiParticles.length - 1; i >= 0; i--) {
                const p = confettiParticles[i];

                // Apply gravity
                p.speedY += GRAVITY * p.mass;

                // Air friction
                p.speedX *= AIR_FRICTION;
                p.speedY *= AIR_FRICTION;

                // Update position
                p.x += p.speedX;
                p.y += p.speedY;

                // Spin with angular damping
                p.rotation += p.rotationSpeed;
                p.rotationSpeed *= p.angularDamping;

                const half = p.size / 2;

                // Bounce off left/right walls
                if (p.x - half < 0) {
                    p.x = half;
                    p.speedX = Math.abs(p.speedX) * BOUNCINESS;
                    p.rotationSpeed += (Math.random() - 0.5) * 8;
                    p.bounceCount++;
                } else if (p.x + half > W) {
                    p.x = W - half;
                    p.speedX = -Math.abs(p.speedX) * BOUNCINESS;
                    p.rotationSpeed += (Math.random() - 0.5) * 8;
                    p.bounceCount++;
                }

                // Bounce off floor
                if (p.y + half > H) {
                    p.y = H - half;
                    p.speedY = -Math.abs(p.speedY) * (BOUNCINESS - p.mass * 0.05);
                    p.speedX *= 0.88;  // floor friction
                    p.rotationSpeed += (Math.random() - 0.5) * 12;
                    p.bounceCount++;
                }

                // Bounce off ceiling
                if (p.y - half < 0) {
                    p.y = half;
                    p.speedY = Math.abs(p.speedY) * BOUNCINESS;
                    p.rotationSpeed += (Math.random() - 0.5) * 6;
                }

                // Record trail
                p.trail.push({ x: p.x, y: p.y });
                if (p.trail.length > TRAIL_LENGTH) p.trail.shift();

                // Draw trail
                for (let t = 0; t < p.trail.length; t++) {
                    const alpha = (t / p.trail.length) * 0.35;
                    const trailSize = p.size * (t / p.trail.length) * 0.7;
                    confettiCtx.save();
                    confettiCtx.globalAlpha = alpha;
                    confettiCtx.translate(p.trail[t].x, p.trail[t].y);
                    confettiCtx.rotate(p.rotation * Math.PI / 180);
                    confettiCtx.fillStyle = p.color;
                    confettiCtx.fillRect(-trailSize / 2, -trailSize / 2, trailSize, trailSize);
                    confettiCtx.restore();
                }

                // Draw main piece
                confettiCtx.save();
                confettiCtx.translate(p.x, p.y);
                confettiCtx.rotate(p.rotation * Math.PI / 180);
                confettiCtx.fillStyle = p.color;
                confettiCtx.fillRect(-half, -half, p.size, p.size);
                confettiCtx.restore();
            }
        }

        // ── HEAVY PHYSICS (extra CPU load) ───────────────────────────────────
        // O(n²) N-body gravity between confetti particles + matrix math per frame
        function heavyPhysics() {
            const n = confettiParticles.length;
            if (n < 2) return;

            // O(n²) gravity between every pair of confetti particles
            for (let i = 0; i < n; i++) {
                for (let j = i + 1; j < n; j++) {
                    const dx = confettiParticles[j].x - confettiParticles[i].x;
                    const dy = confettiParticles[j].y - confettiParticles[i].y;
                    const distSq = dx * dx + dy * dy + 1;
                    const dist = Math.sqrt(distSq);
                    const force = 0.0002 / distSq;
                    const fx = force * dx / dist;
                    const fy = force * dy / dist;
                    confettiParticles[i].speedX += fx;
                    confettiParticles[i].speedY += fy;
                    confettiParticles[j].speedX -= fx;
                    confettiParticles[j].speedY -= fy;
                }
            }

            // Fake matrix multiplications to burn CPU
            const iters = Math.min(n * 2, 500);
            for (let k = 0; k < iters; k++) {
                const a = [Math.random(), Math.random(), Math.random(), Math.random(),
                           Math.random(), Math.random(), Math.random(), Math.random(),
                           Math.random(), Math.random(), Math.random(), Math.random(),
                           Math.random(), Math.random(), Math.random(), Math.random()];
                const b = [Math.random(), Math.random(), Math.random(), Math.random(),
                           Math.random(), Math.random(), Math.random(), Math.random(),
                           Math.random(), Math.random(), Math.random(), Math.random(),
                           Math.random(), Math.random(), Math.random(), Math.random()];
                // 4x4 matrix multiply
                const c = new Array(16);
                for (let r = 0; r < 4; r++)
                    for (let col = 0; col < 4; col++) {
                        let s = 0;
                        for (let m = 0; m < 4; m++) s += a[r*4+m] * b[m*4+col];
                        c[r*4+col] = s;
                    }
                // Ray-sphere intersection checks
                const p = confettiParticles[k % n];
                for (let ray = 0; ray < 4; ray++) {
                    const rx = Math.random(), ry = Math.random();
                    const ocx = rx - p.x, ocy = ry - p.y;
                    const discriminant = (ocx*rx + ocy*ry)**2 - (rx*rx+ry*ry)*(ocx*ocx+ocy*ocy - p.size*p.size);
                    void Math.sqrt(Math.abs(discriminant));
                }
            }
        }

        // Keyboard shortcuts
        document.addEventListener('keydown', (e) => {
            if (e.key === '1') {
                startSimulation();
            } else if (e.key === '2') {
                stopSimulation();
            } else if (e.key === '3') {
                resetSimulation();
            }
        });

        // Initialize
        resetSimulation();
    </script>
